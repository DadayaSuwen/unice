# 绝对化用语合规整改 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 将官网代码兜底、seed 源、线上数据库中的绝对化用语（一流/最先进/全球领先/顶尖/国际领先/世界先进/引领行业）统一替换为合规表述，并产出整改记录文档。

**Architecture:** 三处同一措辞：`app/lib/globals.ts` 兜底与 `payload/seed.ts` seed 源直接文本改写；线上库用新增脚本 `scripts/remediate-absolute-claims.ts` 做「短语级替换」（含富文本 Lexical 文本节点），默认 dry-run、`--apply` 才写库。最终以全库核心禁用词校验 + grep + typecheck 验证。

**Tech Stack:** Next.js 16 / React 19 / Payload CMS 3 (`@payloadcms/db-postgres`) / `tsx` / TypeScript。

## Global Constraints
- 替换措辞必须与设计文档 `docs/superpowers/specs/2026-08-16-绝对化用语整改-design.md` 第 1 节逐字一致；三处（globals / seed / DB）最终字符串一致。
- 保留不动：`国家级新型工业化产业示范基地`、ISO9001、数据性表述、`卓越/高品质/专业/高标准`。
- 富文本（news.content、about.introContent）在库中为 Lexical JSON，只能改 `{type:'text', text}` 节点文本，不改结构。
- 脚本默认 dry-run，只有显式传入 `--apply` 才写库；写库后必须回读校验核心禁用词。
- 运行脚本用 `tsx -r dotenv/config`（直连 `.env` 的 `DATABASE_URL`），不做破坏性操作。

```ts
import 'dotenv/config'
import { getPayload } from 'payload'
import config from '../payload.config'

/**
 * 绝对化用语整改脚本。
 * 默认 dry-run（只读扫描并打印报告）；传入 --apply 才写库并在写库后回读校验核心禁用词。
 * 涉及 Globals 与集合（hero-banners / news / products / careers 等）的全部文本字段，
 * 包括富文本 Lexical 中的 {type:'text', text} 文本节点。

// 短语替换表（核心禁用词 -> 合规词）。按 length 降序在前端排序，保证长短语先被替换。
const REPLACE_MAP: ReadonlyArray<readonly [string, string]> = [
  ['性能指标达到国际领先水平', '性能指标在同类产品中保持较高水平'],
  ['成为全球领先的化工产品供应商，引领行业技术创新和可持续发展', '成为值得信赖的化工产品供应商，持续推动行业技术创新和可持续发展'],
  ['全球领先的化工产品供应商', '值得信赖的化工产品供应商'],
  ['世界一流的生产设施', '高标准的生产设施'],
  ['配备最先进的生产设备和技术', '配备先进的生产设备和技术'],
  ['达到世界先进水平', '符合国家及行业标准'],
  ['汇聚国内外化工领域顶尖专家', '汇聚国内外化工领域资深专家'],
  ['采用一流的研发和生产设备，拥有一流的研发团队', '采用先进的研发和生产设备，拥有专业的研发团队'],
  ['创新化学科技，引领行业未来', '创新化学科技，助力行业发展'],
  ['引领行业技术发展方向', '推动行业技术进步'],
  ['引领行业技术创新和可持续发展', '推动行业技术创新和可持续发展'],
  ['一流的研发和生产设备', '先进的研发和生产设备'],
  ['达到国际领先水平', '达到较高水平'],
  ['国际领先水平', '较高水平'],
  ['世界一流', '高品质'],
  ['一流生产和检测设备', '先进的生产和检测设备'],
  ['全球领先', '行业知名的'],
  ['最先进', '先进'],
  ['世界先进水平', '较高水平'],
  ['顶尖专家', '资深专家'],
  ['拥有一流的研发团队', '拥有专业的研发团队'],
  ['一流产品、提供一流服务', '优质产品、提供专业服务'],
  ['引领行业技术创新', '推动行业技术创新'],
  ['引领行业未来', '助力行业发展'],
  ['引领行业', '推动行业发展'],
  ['一流产品', '优质产品'],
  ['一流服务', '优质服务'],
  ['一流', '优质'],
].sort((a, b) => b[0].length - a[0].length)

// 核心禁用词（写库后全库校验用）
const CORE_BANNED = ['一流', '最先进', '世界一流', '全球领先', '顶尖', '国际领先', '世界先进', '引领行业']

interface ScanResult {
  collection?: string
  id?: number | string
  field: string
  old: string
  new: string
}

// 对任意（可能嵌套的）值递归扫描并就地替换，返回是否命中（dry-run 时不改数据本体，仅报告）。
function walkText(
  value: unknown,
  field: string,
  hits: ScanResult[],
  path: string[] = [],
): { changed: boolean; next: unknown } {
  // 词的顺序都已在 REPLACE_MAP 内定义，此处直接逐层处理
  if (typeof value === 'string') {
    let v = value
    let changed = false
    for (const [old, nw] of REPLACE_MAP) {
      if (v.includes(old)) {
        if (!changed) changed = true
        v = v.split(old).join(nw)
      }
    }
    // 字符串全文替换后与原文比较，优先于“仅报告命中”
    if (changed) {
      hits.push({
        field: [...path, field].join('.'),
        old: value,
        new: v,
      })
      return { changed, next: v }
    }
    return { changed: false, next: value }
  }

  if (Array.isArray(value)) {
    let any = false
    const out = value.map((item, i) => {
      const r = walkText(item, field, hits, [...path, String(i)])
      if (r.changed) any = true
      return r.next
    })
    return { changed: any, next: out }
  }

  if (value && typeof value === 'object') {
    let any = false
    const obj: Record<string, unknown> = {}
    for (const [key, v] of Object.entries(value as Record<string, unknown>)) {
      const r = walkText(v, key, hits, key === 'text' || /^(title|subtitle|description|content|introContent|visionDescription|excerpt|overlayText)$/.test(key) ? path : [...path, key])
      obj[key] = r.next
      if (r.changed) any = true
    }
    return { changed: any, next: obj }
  }

  return { changed: false, next: value }
}

// 写库前剔除 Payload 只读/自管字段，避免 update 时被拒或覆盖版本信息
function stripMeta(value: unknown): unknown {
  if (value && typeof value === 'object' && !Array.isArray(value)) {
    const obj: Record<string, unknown> = { ...(value as Record<string, unknown>) }
    for (const k of ['id', '_id', 'createdAt', 'updatedAt', '_version', '__v']) {
      delete obj[k]
    }
    return obj
  }
  return value
}

async function main() {
  const apply = process.argv.includes('--apply')
  const payload = await getPayload({ config })
  const hits: ScanResult[] = []

  // 全部 Globals
  for (const g of payload.config.globals) {
    const slug = g.slug
    let doc: unknown
    try {
      doc = await payload.findGlobal({ slug: slug as never, depth: 2 })
    } catch {
      // Global 未创建时跳过（seed 前）
      continue
    }
    const r = walkText(doc, slug, hits)
    if (!apply) continue
    if (r.changed) {
      await payload.updateGlobal({ slug: slug as never, data: stripMeta(r.next) as never })
    }
  }

  // 全部集合（只处理存在文本的内容集合）
  const TEXT_COLLECTIONS = ['hero-banners', 'news', 'products', 'careers']
  for (const slug of TEXT_COLLECTIONS) {
    if (!payload.config.collections.some((c: any) => c.slug === slug)) continue
    const { docs } = await payload.find({ collection: slug as never, depth: 2, limit: 5000 })
    for (const d of docs) {
      const r = walkText(d, slug, hits)
      if (!apply) continue
      if (r.changed) {
        await payload.update({ collection: slug as never, id: d.id, data: stripMeta(r.next) as never })
      }
    }
  }

  // 报告
  console.log(`\n=== ${apply ? 'APPLY' : 'DRY-RUN'} 扫描报告（命中 ${hits.length} 处） ===`)
  for (const h of hits) {
    console.log(`\n[${h.collection || ''}${h.id != null ? '#' + h.id : ''} ${h.field}]`)
    console.log(`  旧: ${h.old}`)
    console.log(`  新: ${h.new}`)
  }

  // 写库后校验：重新全库扫描核心禁用词
  if (apply) {
    const remains: string[] = []
    const check = (value: unknown, at: string) => {
      if (typeof value === 'string') {
        for (const w of CORE_BANNED) {
          if (value.includes(w)) remains.push(`${at}(${w}): ${value.slice(0, 60)}`)
        }
      } else if (Array.isArray(value)) {
        value.forEach((v, i) => check(v, `${at}[${i}]`))
      } else if (value && typeof value === 'object') {
        Object.entries(value as Record<string, unknown>).forEach(([k, v]) => check(v, `${at}.${k}`))
      }
    }
    for (const g of payload.config.globals) {
      try { check(await payload.findGlobal({ slug: g.slug as never, depth: 2 }), g.slug) } catch { /* skip */ }
    }
    for (const slug of TEXT_COLLECTIONS) {
      const { docs } = await payload.find({ collection: slug as never, depth: 2, limit: 5000 })
      for (const d of docs) check(d, slug)
    }
    if (remains.length) {
      console.error('\n⚠️ 写库后仍发现核心禁用词残留：')
      remains.forEach((r) => console.error('  ' + r))
      console.error('数据库整改未完全成功，请人工介入。')
      process.exit(1)
    }
    console.log('\n✅ 写库后全库校验通过：核心禁用词无残留。')
  } else {
    console.log('\n（dry-run 模式，未写库。确认无误后执行：`tsx -r dotenv/config scripts/remediate-absolute-claims.ts --apply`）')
  }

  process.exit(0)
}

main().catch((e) => {
  console.error(e)
  process.exit(1)
})
```
